function formatEasyKrwFromUsd(usd,rate=1380){
  const w=Math.round((Number(usd)||0)*(Number(rate)||1380));
  if(w>=1e12){const jo=Math.floor(w/1e12),eok=Math.round((w%1e12)/1e8);return eok?`약 ${jo}조 ${eok.toLocaleString("ko-KR")}억원`:`약 ${jo}조원`;}
  if(w>=1e8){const eok=Math.floor(w/1e8),man=Math.round(((w%1e8)/1e4)/100)*100;return man?`약 ${eok}억 ${man.toLocaleString("ko-KR")}만원`:`약 ${eok}억원`;}
  if(w>=1e4)return`약 ${Math.round(w/1e4).toLocaleString("ko-KR")}만원`;
  return`약 ${w.toLocaleString("ko-KR")}원`;
}
function humanizeMasterMoneyText(text,rate=1380){
  let s=String(text||"");
  s=s.replace(/\$\s*([\d,.]+)\s*([KMB])?\s*[~–-]\s*\$?\s*([\d,.]+)\s*([KMB])?/gi,(m,a,au,b,bu)=>`${formatEasyKrwFromUsd(masterUsdUnit(a,au),rate)}~${formatEasyKrwFromUsd(masterUsdUnit(b,bu),rate).replace(/^약 /,"")}`);
  s=s.replace(/\$\s*([\d,.]+)\s*([KMB])?/gi,(m,a,u)=>formatEasyKrwFromUsd(masterUsdUnit(a,u),rate));
  return s.replace(/약\s+약/g,"약");
}

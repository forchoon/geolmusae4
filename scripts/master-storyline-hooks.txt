  const [masterActivityPerformance,setMasterActivityPerformance]=useState({});
  const [masterArchivedHistory,setMasterArchivedHistory]=useState([]);
  const [masterLegendChartData,setMasterLegendChartData]=useState(null);
  const [masterLegendChartYears,setMasterLegendChartYears]=useState(null);
  const [masterLegendBuyPrice,setMasterLegendBuyPrice]=useState(null);
  const [masterLegendCurrentPrice,setMasterLegendCurrentPrice]=useState(null);
  const [masterLegendLoading,setMasterLegendLoading]=useState(false);
  const [masterLegendResult,setMasterLegendResult]=useState(null);
  const [masterLegendCalcLoading,setMasterLegendCalcLoading]=useState(false);
  const [masterLegendAnimKey,setMasterLegendAnimKey]=useState(0);

  const masterRecentSource=(masterRecentLive&&masterRecentLive[selectedMasterId])||MASTER_RECENT_ACTIVITY[selectedMasterId]||{title:"최근 공개 내역",freshness:"",items:[]};
  const masterRecentKeys=new Set((masterRecentSource.items||[]).map(item=>`${item.ticker||""}|${item.eventDate||""}|${item.badge||""}`));
  const masterHistoryItems=[...masterArchivedHistory,...(MASTER_ACTIVITY_HISTORY[selectedMasterId]||[])]
    .filter((item,idx,arr)=>item&&item.ticker&&item.eventDate&&arr.findIndex(x=>`${x.ticker||""}|${x.eventDate||""}|${x.badge||""}`===`${item.ticker||""}|${item.eventDate||""}|${item.badge||""}`)===idx)
    .filter(item=>!masterRecentKeys.has(`${item.ticker||""}|${item.eventDate||""}|${item.badge||""}`))
    .sort((a,b)=>String(b.eventDate||"").localeCompare(String(a.eventDate||"")));
  const masterActivityItems=masterActivityView==="recent"?(masterRecentSource.items||[]):masterHistoryItems;
  const masterActivityTitle=selectedMasterId==="buffett"?"버핏의 버크셔는 최근 뭘 거래했지?":selectedMasterId==="ark"?"캐시 우드는 최근 뭘 거래했지?":selectedMasterId==="pelosi"?"펠로시 일가는 최근 뭘 거래했지?":`${selectedMaster.shortName||selectedMaster.name}의 공개 기록`;
  const masterLegendRange=masterEvidenceRange(selectedMasterEvidence,selectedMasterCase.year);
  const masterLegendMarket=/\.(KS|KQ)$/.test(selectedMasterCase.ticker)?"kr":"us";
  const masterLegendDisplayPrice=v=>masterLegendMarket==="kr"?formatKRW(v):formatUSD(v);
  const masterLegendReturnPct=masterLegendBuyPrice&&masterLegendCurrentPrice?((masterLegendCurrentPrice/masterLegendBuyPrice-1)*100):null;
  const masterLegendGlossary=(typeof getEvidenceGlossary==="function"?getEvidenceGlossary(selectedMasterEvidence):[]).slice(0,3);

  useEffect(()=>{
    setMasterActivityView("recent");
    setMasterLegendResult(null);
    setMasterQuickAmount(1000);
  },[selectedMasterId]);

  useEffect(()=>{
    let cancelled=false;
    fetch(`/api/master-history?master=${encodeURIComponent(selectedMasterId)}`)
      .then(r=>r.ok?r.json():null)
      .then(data=>{if(!cancelled&&Array.isArray(data?.items))setMasterArchivedHistory(data.items);})
      .catch(()=>{});
    return()=>{cancelled=true;};
  },[selectedMasterId]);

  useEffect(()=>{
    const items=masterRecentSource.items||[];
    if(!items.length)return;
    let cancelled=false;
    const sync=async()=>{
      try{
        await fetch('/api/master-history',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({master:selectedMasterId,items})});
        const r=await fetch(`/api/master-history?master=${encodeURIComponent(selectedMasterId)}`);
        const data=await r.json();
        if(!cancelled&&Array.isArray(data?.items))setMasterArchivedHistory(data.items);
      }catch(e){}
    };
    sync();
    return()=>{cancelled=true;};
  },[selectedMasterId,masterRecentSource.updatedAt,masterRecentSource.freshness]);

  useEffect(()=>{
    let cancelled=false;
    const items=masterActivityItems.filter(item=>/^\d{4}-\d{2}-\d{2}$/.test(item.eventDate||"")&&item.ticker);
    const seed={};items.forEach(item=>{seed[`${item.ticker}|${item.eventDate}`]={loading:true};});
    setMasterActivityPerformance(prev=>({...prev,...seed}));
    Promise.all(items.map(async item=>{
      const key=`${item.ticker}|${item.eventDate}`;
      try{
        const yt=getYahooTicker(item.ticker);
        const[bp,cp]=await Promise.all([fetchYahooPrice(yt,item.eventDate),fetchCurrentPrice(yt)]);
        if(!(bp>0)||!(cp>0))throw new Error("price unavailable");
        return[key,{buyPrice:bp,currentPrice:cp,returnPct:(cp/bp-1)*100,loading:false}];
      }catch(e){return[key,{error:true,loading:false}];}
    })).then(rows=>{
      if(cancelled)return;
      setMasterActivityPerformance(prev=>{const next={...prev};rows.forEach(([k,v])=>next[k]=v);return next;});
    });
    return()=>{cancelled=true;};
  },[selectedMasterId,masterActivityView,masterRecentSource.updatedAt,masterRecentSource.freshness,masterArchivedHistory.length]);

  useEffect(()=>{
    let cancelled=false;
    const load=async()=>{
      setMasterLegendLoading(true);
      setMasterLegendResult(null);
      setMasterLegendChartData(null);
      setMasterLegendChartYears(null);
      setMasterLegendBuyPrice(null);
      setMasterLegendCurrentPrice(null);
      try{
        const yt=getYahooTicker(selectedMasterCase.ticker);
        const exact=/^\d{4}-\d{2}-\d{2}$/.test(selectedMasterEvidence.eventDate||"")?selectedMasterEvidence.eventDate:"";
        const date=exact||getSameDayOfYear(selectedMasterCase.year);
        const[payload,bp,cp]=await Promise.all([fetchChartDataAPI(yt),fetchYahooPrice(yt,date),fetchCurrentPrice(yt)]);
        if(cancelled)return;
        const data=Array.isArray(payload)?payload:payload&&payload.data;
        const years=payload&&payload.years;
        setMasterLegendChartData(data&&data.length?data:getChartData(selectedMasterCase.ticker));
        setMasterLegendChartYears(Array.isArray(years)?years:null);
        setMasterLegendBuyPrice(bp||null);
        setMasterLegendCurrentPrice(cp||null);
      }catch(e){
        if(!cancelled)setMasterLegendChartData(getChartData(selectedMasterCase.ticker));
      }
      if(!cancelled)setMasterLegendLoading(false);
    };
    if(homeMode==="master")load();
    return()=>{cancelled=true;};

  const handleMasterLegendCalculate=async()=>{
    const amount=Number(masterQuickAmount);
    if(masterLegendCalcLoading||!Number.isFinite(amount)||amount<=0)return;
    setMasterLegendCalcLoading(true);
    setMasterLegendResult(null);
    try{
      const yt=getYahooTicker(selectedMasterCase.ticker);
      const exact=/^\d{4}-\d{2}-\d{2}$/.test(selectedMasterEvidence.eventDate||"")?selectedMasterEvidence.eventDate:"";
      const date=exact||getSameDayOfYear(selectedMasterCase.year);
      let bp=masterLegendBuyPrice,cp=masterLegendCurrentPrice;
      if(!(bp>0)||!(cp>0)){[bp,cp]=await Promise.all([fetchYahooPrice(yt,date),fetchCurrentPrice(yt)]);}
      if(!(bp>0)||!(cp>0))throw new Error("price unavailable");
      const principal=amount*10000;
      const ratio=cp/bp;
      const currentValue=principal*ratio;
      const profit=currentValue-principal;
      const returnPct=(ratio-1)*100;
      const exactYears=(new Date()-new Date(date))/(1000*60*60*24*365.25);
      const cagr=(Math.pow(ratio,1/Math.max(exactYears,0.1))-1)*100;
      const sharesCount=Math.floor((masterLegendMarket==="us"?principal/usdToKrw:principal)/bp);
      const resultData={buyPrice:bp,currentPrice:cp,principal,currentValue,profit,returnPct,cagr,years:Math.max(0,Math.floor(exactYears)),isProfit:profit>=0,date,sharesCount,multipleText:formatMultiple(currentValue,principal)};
      setMasterLegendResult(resultData);
      setMasterLegendAnimKey(k=>k+1);
      if(resultData.isProfit){
        if(window.confetti)window.confetti({particleCount:150,spread:70,origin:{y:0.6},colors:[T.accent,"#fbbf24","#ffffff"]});
        if(navigator.vibrate)navigator.vibrate([50,30,50]);
      }else if(navigator.vibrate)navigator.vibrate(200);
      setTimeout(()=>document.getElementById("master-legend-result")?.scrollIntoView({behavior:"smooth",block:"start"}),100);
    }catch(e){
      setMasterLegendResult({error:true});
    }finally{
      setMasterLegendCalcLoading(false);
    }
  };
